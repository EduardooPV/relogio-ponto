  unsigned long lastNowLedConnected = 0;
  bool ledConnected = false;
  int buttonPressed = 1;

  void setup() {
    pinMode(18, OUTPUT);
    pinMode(4, INPUT);
    Serial.begin(115200);
    Serial.println("Hello, ESP32!");
  }

  void loop() {
    unsigned long now = millis();
    int port4 = digitalRead(4);

    if(buttonPressed == 1 && buttonPressed > port4 && ((now - lastNowLedConnected) >= 50)) {
    ledConnected = !ledConnected;

    }


    if (buttonPressed != port4) {
       lastNowLedConnected = now;
    }

      buttonPressed = port4;

   digitalWrite(18, ledConnected);

  }
